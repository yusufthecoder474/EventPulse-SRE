cmake_minimum_required(VERSION 3.16)
project(WirelessLab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(wireless_core analyzer.cpp)
target_include_directories(wireless_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(wireless_core PRIVATE /W4 /WX)
else()
  target_compile_options(wireless_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(wireless_analyzer main.cpp)
target_link_libraries(wireless_analyzer PRIVATE wireless_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(wireless_tests tests.cpp)
  target_link_libraries(wireless_tests PRIVATE wireless_core)
  foreach(case success reject auth_failure disconnect dhcp dns delayed window duplicate out_of_order empty malformed numeric deterministic management ethernet_eapol invalid_transition key_exchange sae multiple_clients reconnect)
    add_test(NAME wireless_${case} COMMAND wireless_tests ${case})
  endforeach()
endif()
